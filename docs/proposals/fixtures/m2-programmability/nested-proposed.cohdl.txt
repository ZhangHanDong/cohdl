// Synthetic check-only interfaces; no procurement parts or hardware validation.
device SignalSource {
    pins { required OUT: 1 [output] }
}
device SignalSink {
    pins { required IN: 1 [input] }
}
device Ground {
    pins { required GND: 1 [passive] }
}
device SeriesR<R: Resistance> {
    spec { resistance: R }
    pins {
        required A: 1 [passive]
        required B: 2 [passive]
    }
}
device ShuntC<C: Capacitance> {
    spec { capacitance: C }
    pins {
        required A: 1 [passive]
        required B: 2 [passive]
    }
}

fn rc_channel<R: Resistance, C: Capacitance>(src: Pin, dst: Pin, gnd: Pin) {
    inst r: SeriesR<R>
    inst c: ShuntC<C>
    net IN: src, r.A
    net OUT: r.B, c.A, dst
    net GND [gnd]: gnd, c.B
}

// PROPOSED M2 syntax, not implemented.
fn filter_bank<const N: Int, R: Resistance, C: Capacitance>(src: Pin, gnd: Pin) {
    inst receivers: [SignalSink; N]
    for channels: i in 0..N {
        rc_channel::<R, C>(src, receivers[i].IN, gnd)
    }
}
fn receiver_cluster<const N: Int, R: Resistance, C: Capacitance>(src: Pin, gnd: Pin) {
    filter_bank::<N, R, C>(src, gnd)
}
design NestedBank {
    inst source: SignalSource
    inst ground: Ground
    receiver_cluster::<3, 1kohm, 100nF>(source.OUT, ground.GND)
}
